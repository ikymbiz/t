' ── ファイル保存 ──────────────────────────
Sub saveData()
  Dim fso, f, path, data
  Set fso = CreateObject("Scripting.FileSystemObject")

  path = InputBox("保存先のファイルパスを入力してください（例: C:\Users\user\PPTX点検.json）", "保存", document.getElementById("currentFile").innerText)
  If path = "" Then Exit Sub
  If Right(path,5) <> ".json" Then path = path & ".json"

  data = window.external.getData()  ' JSから取得
  Set f = fso.OpenTextFile(path, 2, True, -1)
  f.Write data
  f.Close

  document.getElementById("currentFile").innerText = path
  setStatus "保存しました: " & path
End Sub

' ── ファイル読み込み ──────────────────────────
Sub loadData()
  Dim fso, f, path, data
  Set fso = CreateObject("Scripting.FileSystemObject")

  path = InputBox("読み込むファイルのパスを入力してください", "読み込み", "")
  If path = "" Then Exit Sub
  If Not fso.FileExists(path) Then
    MsgBox "ファイルが見つかりません: " & path, 16, "エラー"
    Exit Sub
  End If

  Set f = fso.OpenTextFile(path, 1, False, -1)
  data = f.ReadAll
  f.Close

  Call window.external.setData(data)  ' JSへ渡す
  document.getElementById("currentFile").innerText = path
  setStatus "読み込みました: " & path
End Sub

' ── レポートをHTMLで保存 ──────────────────────────
Sub exportReportFile(html)
  Dim fso, f, path
  Set fso = CreateObject("Scripting.FileSystemObject")

  path = InputBox("レポートの保存先を入力してください（例: C:\Users\user\点検レポート.html）", "レポート出力", "")
  If path = "" Then Exit Sub
  If InStr(path, ".htm") = 0 Then path = path & ".html"

  Set f = fso.OpenTextFile(path, 2, True, -1)
  f.Write html
  f.Close

  setStatus "レポートを出力しました: " & path
  If MsgBox("レポートを開きますか？", 4, "確認") = 6 Then
    CreateObject("WScript.Shell").Run path
  End If
End Sub

Sub setStatus(msg)
  document.getElementById("statusMsg").innerText = msg
End Sub
</script>

var currentTab = 0;
function switchTab(n) {
  document.querySelectorAll('.panel').forEach(function(p,i){
    p.className = 'panel' + (i===n?' active':'');
  });
  document.querySelectorAll('.tab').forEach(function(t,i){
    t.className = 'tab' + (i===n?' active':'');
  });
  currentTab = n;
  if(n===3) renderSummary();
  if(n===4) renderReport();
  if(n===5) { document.getElementById('promptOutputCard').style.display='none'; document.getElementById('copyBtn').style.display='none'; }
}

function getCheckedMode() {
  var radios = document.getElementsByName('promptMode');
  for(var i=0;i<radios.length;i++) if(radios[i].checked) return radios[i].value;
  return 'full';
}

function generatePrompt() {
  var mode  = getCheckedMode();
  var scope = document.getElementById('slideScope').value.trim() || '全スライド';

  // ── A-1 表記揺れリスト ──
  var wordingRows = DATA.wording.map(function(r){
    return '  - 正式表記「' + r.correct + '」← 禁止：' + r.wrong;
  }).join('\n') || '  （未登録）';

  // ── A-2 禁止キーワード ──
  var bannedRows = '';
  var bannedCats = ["社外秘・機密","不適切・品位を損なう表現","過度な断言表現","作業中痕跡"];
  bannedCats.forEach(function(cat){
    var items = DATA.banned.filter(function(r){return r.cat===cat;});
    if(items.length){
      bannedRows += '  【'+cat+'】\n';
      items.forEach(function(r){ bannedRows += '    - '+r.kw+'\n'; });
    }
  });
  if(!bannedRows) bannedRows = '  （未登録）\n';

  // ── A-3 禁止オブジェクト ──
  var objRows = '';
  var objCats = ["禁止オブジェクト","禁止デザイン"];
  objCats.forEach(function(cat){
    var items = DATA.objects.filter(function(r){return r.cat===cat;});
    if(items.length){
      objRows += '  【'+cat+'】\n';
      items.forEach(function(r){ objRows += '    - '+r.item+'\n'; });
    }
  });
  if(!objRows) objRows = '  （未登録）\n';

  // ── NG・要確認リスト（再点検モード用） ──
  var ngList = '';
  var ngCount = 0;
  CHECKLIST.forEach(function(sec){
    sec.items.forEach(function(item,i){
      var k=sec.id+'-'+i;
      var st=DATA.statuses[k];
      if(st==='ng'||st==='warn'){
        ngCount++;
        var loc = DATA.locs[k] ? '【'+DATA.locs[k]+'】' : '';
        var before = DATA.befores[k] ? '\n    原文：「'+DATA.befores[k]+'」' : '';
        var after  = DATA.afters[k]  ? '\n    修正案：「'+DATA.afters[k]+'」' : '';
        ngList += '  No.'+ngCount+' ['+sec.id+'-'+(i+1)+'] '+(st==='ng'?'❌':'⚠️')+' '+item+'\n';
        if(loc||before||after) ngList += '    '+loc+before+after+'\n';
      }
    });
  });

  // ── プロンプト本文組み立て ──
  var p = '';

  // 共通ヘッダー
  p += '# PowerPoint スライド 形式点検依頼\n\n';
  p += '添付のPowerPointスライドを、以下の基準に従って点検してください。\n';
  p += '対象範囲：' + scope + '\n\n';

  p += '════════════════════════════════════\n';
  p += '【カスタマイズ基準（必ず遵守してください）】\n';
  p += '════════════════════════════════════\n\n';

  p += '▼ A-1 表記揺れ注意ワード（以下の正式表記に統一してください）\n';
  p += wordingRows + '\n\n';

  p += '▼ A-2 使用禁止キーワード・表現（これらが含まれていればNGとして報告してください）\n';
  p += bannedRows + '\n';

  p += '▼ A-3 使用禁止オブジェクト・デザイン要素（これらが含まれていればNGとして報告してください）\n';
  p += objRows + '\n';

  p += '════════════════════════════════════\n';
  p += '【点検内容】\n';
  p += '════════════════════════════════════\n\n';

  if(mode === 'ng') {
    // 再点検モード
    p += '以下は前回の点検でNG・要確認と判定された項目です。\n';
    p += '修正後のスライドが正しく修正されているか確認してください。\n\n';

var openSecs = {1:true};
function renderChecklist() {
  var c = document.getElementById('checklistContainer');

  CHECKLIST.forEach(function(sec) {
    var ngCnt=0, warnCnt=0;
    sec.items.forEach(function(_,i){

      if(DATA.statuses[k]==='ng')   ngCnt++;
      if(DATA.statuses[k]==='warn') warnCnt++;
    });
    var badges = (ngCnt?'<span style="color:#dc2626;font-weight:700;font-size:11px;margin-left:6px">❌ '+ngCnt+'</span>':'') +
                 (warnCnt?'<span style="color:#d97706;font-weight:700;font-size:11px;margin-left:4px">⚠️ '+warnCnt+'</span>':'');

    var card = document.createElement('div');
    card.className = 'card';
    var isOpen = openSecs[sec.id];

    var itemsHtml = sec.items.map(function(item, i) {
      var k = sec.id+'-'+i;
      var st = DATA.statuses[k] || 'none';
      var showDiff = (st==='warn'||st==='ng') ? 'show' : '';
      return '<div class="checklist-item">' +
        '<div class="item-top">' +
          '<div class="item-no">' + sec.id + '-' + (i+1) + '</div>' +
          '<div style="flex:1">' +
            '<div class="item-text">' + item + '</div>' +
            '<div class="item-buttons">' +
              makeStatusBtn(k,'ok',st) + makeStatusBtn(k,'warn',st) + makeStatusBtn(k,'ng',st) +
            '</div>' +
            '<div class="diff-box ' + showDiff + '" id="diff_'+k+'">' +
              '<div class="diff-row">' +
                '<div class="diff-col"><div class="diff-label before">🔴 原文</div>' +
                  '<textarea rows="2" onchange="DATA.befores[\''+k+'\']=this.value" placeholder="問題のある原文を貼り付け" style="border-color:#fca5a5">'+esc(DATA.befores[k]||'')+'</textarea>' +
                '</div>' +
                '<div class="diff-col"><div class="diff-label after">✅ 修正案</div>' +
                  '<textarea rows="2" onchange="DATA.afters[\''+k+'\']=this.value" placeholder="修正後の文章・数値" style="border-color:#86efac">'+esc(DATA.afters[k]||'')+'</textarea>' +

                '</div>' +
              '</div>' +
            '</div>' +
          '</div>' +
        '</div>' +
      '</div>';

function makeStatusBtn(k, s, current) {
  var labels = {ok:'✅ OK', warn:'⚠️ 要確認', ng:'❌ NG'};
  var act = current===s ? ' active' : '';
  return '<button class="status-btn '+s+act+'" onclick="setStatus(\''+k+'\',\''+s+'\')">' + labels[s] + '</button>';
}
function setStatus(k, s) {
  DATA.statuses[k] = DATA.statuses[k]===s ? 'none' : s;
  renderChecklist();
  renderSummary();
}
function toggleSec(id) {
  openSecs[id] = !openSecs[id];
  document.getElementById('secbody'+id).style.display = openSecs[id] ? 'block' : 'none';
  var a = document.getElementById('arrow'+id);
  a.className = 'arrow' + (openSecs[id]?' open':'');
}

function renderReport() {
  var c = document.getElementById('reportContainer');
  var issues = [];
  CHECKLIST.forEach(function(sec){
    sec.items.forEach(function(item,i){
      var k = sec.id+'-'+i;
      var st = DATA.statuses[k];
      if(st==='ng'||st==='warn'){
        issues.push({ no:'No.'+(issues.length+1<10?'0':'')+(issues.length+1), sec:sec.id, item:item, st:st,
          before:DATA.befores[k]||'', after:DATA.afters[k]||'', loc:DATA.locs[k]||'' });
      }
    });
  });

  if(!issues.length){
    c.innerHTML = '<div class="card"><div class="card-body" style="color:#16a34a;font-weight:700;padding:20px">🎉 NG・要確認の項目はありません。</div></div>';
    return;
  }

  var html = issues.map(function(iss,idx) {
    var stLabel = iss.st==='ng' ? '❌ NG' : '⚠️ 要確認';
    var stColor = iss.st==='ng' ? '#dc2626' : '#d97706';
    var stBg    = iss.st==='ng' ? '#fee2e2' : '#fef9c3';

      '<div class="card-body">' +

          '<span style="font-weight:900;font-size:13px;color:'+stColor+'">['+(idx+1<10?'0'+(idx+1):idx+1)+'] '+stLabel+'</span>' +
          '<span style="font-size:12px;color:#6b7280">項目 ' + iss.sec + '</span>' +
        '</div>' +
        '<div style="font-weight:700;margin-bottom:10px;font-size:13px">'+iss.item+'</div>' +
        (iss.loc ? '<div style="margin-bottom:8px"><span class="diff-label loc">📍 該当箇所</span> <span style="font-size:12px">'+esc2(iss.loc)+'</span></div>' : '') +
        '<table style="width:100%;border:1px solid #e5e7eb;border-radius:8px;overflow:hidden;border-collapse:collapse">' +
          '<tr><td style="width:90px;padding:10px 12px;background:#fee2e2;font-weight:700;font-size:12px;color:#dc2626;vertical-align:top">🔴 原文</td>' +
               '<td style="padding:10px 12px;font-size:13px;white-space:pre-wrap;background:#fff">' + (esc2(iss.before)||'<span style="color:#9ca3af">（未記入）</span>') + '</td></tr>' +

        '</table>' +
      '</div></div>';
  }).join('');

  c.innerHTML = '<div style="margin-bottom:12px;font-weight:700;font-size:13px;color:#374151">指摘件数：' + issues.length + ' 件</div>' + html;
